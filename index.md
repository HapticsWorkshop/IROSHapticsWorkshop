<section class="wk-section">
  <h2 class="wk-section__title">About the Workshop</h2>

  <p class="wk-section__lead">
    Human manipulation relies fundamentally on rich tactile and kinesthetic
    feedback to regulate forces, adapt to contact conditions, and achieve fine
    motor control. Robotic manipulation, in contrast, still leans predominantly
    on vision and open-loop strategies &mdash; limiting robustness in
    contact-rich, uncertain environments. This workshop asks how recent progress
    in tactile sensing, haptic interfaces, and contact modeling can be
    integrated into modern manipulation frameworks, and brings together
    perspectives from robotics, haptics, human&ndash;robot interaction, and
    industry to promote a unified approach toward haptic-enabled dexterity.
  </p>

  <a class="wk-more" href="{{ '/about/' | relative_url }}">Read the full motivation</a>
</section>

      <div class="wk-card__num">Goal 01</div>
      <div class="wk-card__title">Surface what is now possible</div>
      <p class="wk-card__body">
        Highlight recent advances in tactile sensing, haptic actuation, and
        interface design that measurably improve manipulation in robotic and
        human&ndash;machine systems.
      </p>

    <div class="wk-card">
      <div class="wk-card__num">Goal 02</div>
      <div class="wk-card__title">Name the real bottlenecks</div>
      <p class="wk-card__body">
        Identify the technical challenges and research gaps in tactile sensing,
        contact modeling, and haptic rendering that currently cap manipulation
        performance.
      </p>
    </div>

    <div class="wk-card">
      <div class="wk-card__num">Goal 03</div>
      <div class="wk-card__title">Connect fragmented communities</div>
      <p class="wk-card__body">
        Promote interdisciplinary dialogue among researchers working across
        robotics, haptics, tactile sensing, and human perception, whose efforts
        rarely meet in one room.
      </p>
    </div>

    <div class="wk-card">
      <div class="wk-card__num">Goal 04</div>
      <div class="wk-card__title">Push toward translation</div>
      <p class="wk-card__body">
        Discuss emerging application domains for haptic-enabled dexterity:
        teleoperation, advanced manufacturing, assistive robotics, and
        rehabilitation technologies.
      </p>
    </div>
  </div>
</section>

// Live countdown in the hero. Falls through three states so the pill stays
// accurate without anyone editing it: submissions open -> workshop upcoming -> past.
(function () {
  var POSTER_DEADLINE = new Date("2026-08-14T23:59:59-12:00"); // anywhere on Earth
  var WORKSHOP_START = new Date("2026-10-01T08:00:00-04:00");  // Pittsburgh, EDT

  var pill = document.getElementById("wk-countdown");
  var text = document.getElementById("wk-countdown-text");
  if (!pill || !text) return;

  var now = new Date();
  var days = function (target) {
    return Math.ceil((target - now) / 86400000);
  };
  var plural = function (n) {
    return n === 1 ? "day" : "days";
  };

  var message;
  if (now < POSTER_DEADLINE) {
    var d = days(POSTER_DEADLINE);
    message =
      d <= 1
        ? "Poster submissions close today"
        : "Poster submissions close in " + d + " " + plural(d);
  } else if (now < WORKSHOP_START) {
    var w = days(WORKSHOP_START);
    message = w <= 1 ? "The workshop is tomorrow" : "Workshop in " + w + " " + plural(w);
  } else {
    message = "Thank you to everyone who joined us at IROS 2026";
  }

  text.textContent = message;
  pill.hidden = false;
})();
</script>
